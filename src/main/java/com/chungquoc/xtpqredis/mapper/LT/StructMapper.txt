import org.mapstruct.Mapper;
import org.mapstruct.factory.Mappers;

@Mapper(componentModel = "spring") // để Spring tự inject
public interface UserMapper {
    UserMapper INSTANCE = Mappers.getMapper(UserMapper.class);

    UserDto toDto(User user);
    User toEntity(UserDto userDto);
}


@Service
public class UserService {

    @Autowired
    private UserMapper userMapper;

    public UserDto getUser(String username) {
        User user = userRepository.findByUsername(username); // giả định bạn có userRepository
        return userMapper.toDto(user);
    }
}

@Getter
@Setter
class User {
    private String username;
    private String password;
    private int age;
}


@Getter
@Setter
class UserDto {
    private String hoTen;
    private int tuoi;
}



@Mapper(componentModel = "spring")
public interface UserMapper {
    UserMapper INSTANCE = Mappers.getMapper(UserMapper.class);

    @Mapping(source = "username", target = "hoTen")
    @Mapping(source = "age", target = "tuoi")
    UserDto toDto(User user);

    @Mapping(source = "hoTen", target = "username")
    @Mapping(source = "tuoi", target = "age")
    User toEntity(UserDto userDto);
}



@Service
public class UserService {

    @Autowired
    private UserMapper userMapper;

    public UserDto getUserDto(String username) {
        User user = userRepository.findByUsername(username);
        return userMapper.toDto(user);
    }
}
