+ Bước 1: Nghĩ ra một cái validate để dùng luôn :

  @Phone()

+ Bước 2: Tạo ra một cái anotation, sau đó thêm các anotation vào cho nó:
  public @interface Phone {

  }


  @Documented - Chỉ định rằng anotation sẽ xuất hiện trong javadoc của field, class, method
  sử dụng nó.

  @Constraint(validatedBy = PhoneValidator.class)-Xác định class nào sẽ thực hiện
  việc kiểm tra hợp lệ này.

  @Target({ FIELD }) - cho biết anotation này sẽ được dùng ở đâu
  @Retention(RUNTIME) - cho biết anotation này sẽ đươc giữ lại khi chạy chương trình



 @Documented
 @Constraint(validatedBy = PhoneValidator.class)
 @Target({ FIELD })
 @Retention(RUNTIME)
 public @interface Phone {
    String message() default "Số điện thoại không hợp lệ";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
}

+ Viết class thực hiện việc kiểm tra ngoại lệ:


public class PhoneValidator implements ConstraintValidator<Phone, String> {

    private static final String PHONE_PATTERN = "^(0|\\+84)[0-9]{9}$";

    @Override
    public boolean isValid(String phone, ConstraintValidatorContext context) {
        if (phone == null) return false;
        return phone.matches(PHONE_PATTERN);
    }
}

public class PhoneValidator implements ConstraintValidator<Phone,String> {
  @Override
  public void initialize(Phone phone) {
    ConstraintValidator.super.initialize(phone);
  }

  @Override
  public boolean isValid(String phone, ConstraintValidatorContext context) {
    if (phone == null || phone.isBlank()) {
      return false;
    }
    String[] validPatterns = {
        "^\\d{10}$",                      // 10 số: 0387654321
        "^\\+84\\d{9}$",                  // +84: +84387654321
        "^\\d{3}-\\d{3}-\\d{4}$",         // gạch ngang: 038-765-4321
        "^\\d{3} \\d{3} \\d{4}$",         // cách: 038 765 4321
    };

    for (String pattern : validPatterns) {
      if (phone.matches(pattern)) {
        return true;
      }
    }

    return false;
  }

}
